Climate Suitability Map Generator

This notebook generates the interactive map layers for the UBC Botanical Garden Climate Adaptation Analysis.

Methodology: To ensure performance on the web (GitHub Pages) with ~48,000 data points, this notebook splits the visualization into 8 separate HTML files (2 Life Forms × 4 Climate Scenarios).

Input: ../data/FullJoin3_with_climate_ratings-proofed.csv
Output: Interactive HTML maps saved to ../docs/maps/

import pandas as pd
import numpy as np
import plotly.graph_objects as go
import plotly.io as pio
import os

# Ensure the output directory exists for the website build
# We save directly to ../docs/maps so the website can serve them
output_dir = "../docs/maps"
os.makedirs(output_dir, exist_ok=True)

# Disable default renderer to prevent notebook bloat during Quarto render
pio.renderers.default = None

1. Load & Process Data

Standardizing coordinates, applying jitter for visibility, and formatting the dataframe.

# 1. Load Data
df = pd.read_csv('../data/FullJoin3_with_climate_ratings-proofed.csv')

# 2. Rename columns for clarity
rename_dict = {
    'climate_rating_current_sheet': 'Current',
    'climate_rating_emissions_limited_2050': '2050',
    'climate_rating_business_as_usual_2090': '2090',
    'climate_rating_bau_plus_1degree_2090': '2090_plus_1_degree',
    'GenusSpecies': 'Taxon',
}

# 3. Standardize Coordinates & Filter
# Note: Logic preserved from original notebook (X=Lat, Y=Long based on min/max logic)
df['LocationCoordY_fixed'] = df[['LocationCoordX', 'LocationCoordY']].min(axis=1) # Longitude
df['LocationCoordX_fixed'] = df[['LocationCoordX', 'LocationCoordY']].max(axis=1) # Latitude

exclude_locs = ['Nursery', 'Nitobe Memorial Garden', 'Nitobe']
df_clean = df[~df['LocationName'].isin(exclude_locs)].copy()
df_clean = df_clean[df_clean['LocationCoordY_fixed'] < -123.245].copy()

# 4. Apply Jitter to prevent dot overlap
jitter_strength = 0.00006 
np.random.seed(42) 
df_clean['LocationCoordX_Jittered'] = df_clean['LocationCoordX_fixed'] + np.random.uniform(-jitter_strength, jitter_strength, size=len(df_clean))
df_clean['LocationCoordY_Jittered'] = df_clean['LocationCoordY_fixed'] + np.random.uniform(-jitter_strength, jitter_strength, size=len(df_clean))

# 5. Reshape Data
df_ready = df_clean.rename(columns=rename_dict)
cols_to_keep = [
    'ItemAccNoFull', 'LocationCoordX_Jittered', 'LocationCoordY_Jittered', 
    'LifeForm', 'Taxon', 
    'Current', '2050', '2090', '2090_plus_1_degree'
]
final_merged_df = df_ready[cols_to_keep].copy()

# Map to broader LifeForm categories
lifeform_mapping = {
    'Shrub': 'Woody', 'Tree': 'Woody', 'Shrub or Tree': 'Woody',
    'Climber_Liana_Vine': 'Woody', 'Herbaceous Perennial': 'Perennial',
    'Bulb, Corm, or Tuber': 'Perennial', 'Annual': 'Short-lived',
    'Biennial': 'Short-lived', 'Habit unknown': 'Unknown'
}
final_merged_df['LifeForm'] = final_merged_df['LifeForm'].map(lifeform_mapping).fillna('Unknown')

# Melt to long format for plotting
long_format_df = pd.melt(
    final_merged_df,
    id_vars=['ItemAccNoFull', 'LocationCoordX_Jittered', 'LocationCoordY_Jittered', 'Taxon', 'LifeForm'],
    value_vars=['Current', '2050', '2090', '2090_plus_1_degree'],
    var_name='Era',
    value_name='ClimateRating'
).dropna(subset=['ClimateRating'])

# 6. Final Filter
target_lifeforms = ['Woody', 'Perennial']
plot_df = long_format_df[long_format_df['LifeForm'].isin(target_lifeforms)].copy()

print(f"Data Ready: {len(plot_df)} points (Full Dataset - No Downsampling).")
Data Ready: 47904 points (Full Dataset - No Downsampling).

2. Generate Interactive Maps

This loop generates 8 separate HTML files. It uses go.Scattermapbox (WebGL) for performance.

# --- CONFIGURATION ---
color_map = {
    1: '#2ecc71', # Suitable (Green)
    2: '#f1c40f', # Caution (Yellow)
    3: '#e67e22', # Vulnerable (Orange)
    4: '#e74c3c', # Unsuitable (Red)
    5: '#95a5a6'  # Unknown (Grey)
}

# Era labels for nicer titles
era_labels = {
    'Current': 'Current Conditions',
    '2050': '2050 (Emissions Limited)',
    '2090': '2090 (Business as Usual)',
    '2090_plus_1_degree': '2090 (+1°C Warming)'
}

# --- GENERATION LOOP ---
print("Generating maps...")

for lifeform in ['Woody', 'Perennial']:
    for era in ['Current', '2050', '2090', '2090_plus_1_degree']:
        
        # 1. Filter Data for this specific scenario
        subset = plot_df[(plot_df['LifeForm'] == lifeform) & (plot_df['Era'] == era)]
        
        # 2. Calculate center for this specific subset
        mid_lat = subset['LocationCoordX_Jittered'].mean()
        mid_lon = subset['LocationCoordY_Jittered'].mean()

        # 3. Initialize Figure
        fig = go.Figure()

        # 4. Add Traces by Rating (Sorted for legend consistency)
        for rating in sorted(subset['ClimateRating'].unique()):
            rating_int = int(rating) # Ensure integer for color map lookup
            rating_subset = subset[subset['ClimateRating'] == rating]
            
            # Use Scattermapbox for high-performance WebGL rendering
            fig.add_trace(
                go.Scattermapbox(
                    lat=rating_subset['LocationCoordX_Jittered'],
                    lon=rating_subset['LocationCoordY_Jittered'],
                    mode='markers',
                    marker=go.scattermapbox.Marker(
                        size=6,
                        color=color_map.get(rating_int, 'black'),
                        opacity=0.8
                    ),
                    name=f"Rating {rating_int}",
                    text=rating_subset['Taxon'], # Hover text
                    hoverinfo='text+name'
                )
            )

        # 5. Configure Layout
        fig.update_layout(
            mapbox=dict(
                style="carto-positron",
                center=dict(lat=mid_lat, lon=mid_lon),
                zoom=14.5
            ),
            title_text=f"{lifeform} Plants - {era_labels[era]}",
            height=800,
            margin={"r":0,"t":40,"l":0,"b":0},
            legend=dict(title="Suitability Rating")
        )

        # 6. Save as separate HTML file
        # 'include_plotlyjs="cdn"' keeps file size small (~500KB instead of 3MB)
        filename = f"{lifeform}_{era}.html"
        filepath = os.path.join(output_dir, filename)
        
        fig.write_html(filepath, include_plotlyjs='cdn')
        print(f"  ✓ Saved: {filepath} ({len(subset)} points)")

print("\nAll interactive maps generated successfully.")
